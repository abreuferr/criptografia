\section{Algoritmos}

\subsection{Chave Simétrica}
	\begin{itemize}
		\item Algoritmo - AES (Advanced Encryption Standard), com chave de 256 bits;
		\item Modo recomendado - GCM, uma cifra autenticada (AEAD) que protege a confidencialidade e permite detectar alterações no texto cifrado;
		\item Nonce - preferencialmente 96 bits e nunca reutilizado com a mesma chave;
		\item Tag de autenticação - 128 bits;
		\item Referências - \href{https://nvlpubs.nist.gov/nistpubs/FIPS/NIST.FIPS.197.pdf}{FIPS PUB 197} (AES) e \href{https://csrc.nist.gov/pubs/sp/800/38/d/final}{NIST SP 800-38D} (GCM).
	\end{itemize}

O modo CBC pode ser necessário para compatibilidade com sistemas legados, mas fornece apenas confidencialidade. Nesse caso, use um IV imprevisível e autentique o IV e o texto cifrado com um MAC, usando uma chave independente; verifique o MAC antes de decifrar.

\subsection{Chave Assimétrica}
	\begin{itemize}
		\item Algoritmo - RSA
		\item Tamanho Mínimo da chave - 2048 bits
		\item Recomendação - NIST SP 800-57 Part 1, Rev. 5 (January 2020)
	\end{itemize}

\subsection{Hash}
	\begin{itemize}
		\item Algoritmo - SHA-2, especificamente SHA-256, SHA-384, e SHA-512
		\item Tamanho da chave - 256/512 bits
		\item Recomendação - \href{https://nvlpubs.nist.gov/nistpubs/FIPS/NIST.FIPS.180-4.pdf}{NIST - FIPS PUB 180-4}
	\end{itemize}

\subsection{HMAC}
	\begin{itemize}
		\item Algoritmo - SHA-2, especificamente SHA-256, SHA-384, e SHA-512
		\item Tamanho da chave - 256/512 bits
		\item Recomendação - \href{https://csrc.nist.gov/pubs/sp/800/224/ipd}{NIST SP 800-224}
	\end{itemize}

\subsection{Key Derivation Function (KDF)}
	\begin{itemize}
		\item Algoritmo -  \href{https://www.rfc-editor.org/rfc/rfc8018.html}{PBKDF2}
		\item Características do Algoritmo
		\begin{itemize}
			\item \textbf{Número de Iterações} : 600.000 interações;
			\item \textbf{Tamanho do Salt} : O salt deve ter pelo menos 128 bits (16 bytes) de comprimento;
			\item \textbf{Tamanho da chave} : 256 bits;
		\end{itemize}
		\item Recomendação - \href{https://csrc.nist.gov/publications/detail/sp/800-132/final}{NIST Special Publication 800-132};
	\end{itemize}
